rules_version = '2';
service cloud.firestore {
  match /databases/{database}/documents {
    function authed() { return request.auth != null; }
    function inList() { return authed() && exists(/databases/$(database)/documents/allowlist/$(request.auth.uid)); }
    function myRole() { return get(/databases/$(database)/documents/allowlist/$(request.auth.uid)).data.role; }
    function isBoss() { return inList() && (myRole() == 'owner' || myRole() == 'admin'); }

    match /public/{doc} { allow read: if true; allow write: if isBoss(); }
    match /inbox/{doc} { allow create: if true; allow read, delete: if isBoss(); }
    match /data/{doc} { allow read, write: if inList(); }
    match /settings/{doc} { allow read: if inList(); allow write: if isBoss(); }
    match /allowlist/{uid} {
      allow read: if authed() && (request.auth.uid == uid || isBoss());
      allow create, update, delete: if isBoss();
    }
    match /{document=**} { allow read, write: if false; }
  }
}
