        const format = (n) => Number(n || 0).toLocaleString('ru-RU');
        const today = new Date();
        document.getElementById('date').textContent = 'от ' + today.toLocaleDateString('ru-RU', { day: 'numeric', month: 'long', year: 'numeric' });

        function loadReport() {
            const params = new URLSearchParams(window.location.search);
            const reportId = params.get('id');
            
            if (!reportId) {
                document.getElementById('loading').style.display = 'none';
                document.getElementById('error').style.display = 'block';
                document.getElementById('error').textContent = '❌ Не указан ID отчёта в ссылке.';
                return;
            }

            db.collection('short_reports').doc(reportId).get().then(doc => {
                document.getElementById('loading').style.display = 'none';
                if (!doc.exists) {
                    document.getElementById('error').style.display = 'block';
                    document.getElementById('error').textContent = '❌ Отчёт не найден или был удалён.';
                    return;
                }
                
                const r = doc.data();
                const collected = Number(r.collected) || 0;
                const spent = Number(r.spent) || 0;
                const card = Number(r.card) || 0;
                const cash = Number(r.cash) || 0;
                const total = collected - spent + card + cash;

                document.getElementById('out-title').textContent = '📊 ' + (r.title || 'Краткий отчёт');
                document.getElementById('out-collected').textContent = format(collected) + ' ₽';
                document.getElementById('out-spent').textContent = '−' + format(spent) + ' ₽';
                document.getElementById('out-card').textContent = '+' + format(card) + ' ₽';
                document.getElementById('out-cash').textContent = '+' + format(cash) + ' ₽';
                document.getElementById('out-total').textContent = format(total) + ' ₽';

                document.getElementById('steps').innerHTML = `
                    <p>1. Взносы: <b>${format(collected)} ₽</b></p>
                    <p>2. Минус расходы: ${format(collected)} − ${format(spent)} = <b>${format(collected - spent)} ₽</b></p>
                    <p>3. Плюс карта: ${format(collected - spent)} + ${format(card)} = <b>${format(collected - spent + card)} ₽</b></p>
                    <p>4. Плюс наличные: ${format(collected - spent + card)} + ${format(cash)} = <b>${format(total)} ₽</b></p>
                    <p class="result">💡 Итоговый баланс: ${format(total)} ₽</p>
                `;

                if (r.comment) {
                    document.getElementById('comment-box').style.display = 'block';
                    document.getElementById('comment-box').textContent = '💬 ' + r.comment;
                }

                document.getElementById('report').style.display = 'block';
            }).catch(err => {
                document.getElementById('loading').style.display = 'none';
                document.getElementById('error').style.display = 'block';
                document.getElementById('error').textContent = '❌ Ошибка загрузки: ' + err.message;
            });
        }

        function downloadPDF() {
            const element = document.getElementById('report');
            const title = document.getElementById('out-title').textContent.replace(/[^a-zа-яё0-9]/gi, '_');
            
            const opt = {
                margin: 10,
                filename: `Отчет_${title}.pdf`,
                image: { type: 'jpeg', quality: 0.98 },
                html2canvas: { scale: 2, backgroundColor: '#ffffff' },
                jsPDF: { unit: 'mm', format: 'a4', orientation: 'portrait' }
            };

            html2pdf().set(opt).from(element).save();
        }

        loadReport();
