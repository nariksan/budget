rules_version = '2';
service cloud.firestore {
  match /databases/{database}/documents {
    
    // === ФОТОАЛЬБОМЫ (НОВОЕ) ===
    match /albums/{albumId} {
      // Читать альбомы могут только авторизованные пользователи (родители/админы/владелец)
      allow read: if request.auth != null;
      // Создавать и удалять альбомы могут авторизованные пользователи (через админку владельца или админа)
      allow create, delete: if request.auth != null;
    }
    
    // === СООБЩЕНИЯ ===
    // 💡 Подсказка: если вы хотите, чтобы гости (без входа) могли отправлять сообщения 
    // через кнопку "Помощь и доступ", замените 'request.auth != null' на 'true' в строке allow create.
    match /messages/{messageId} {
      allow create: if request.auth != null;
      allow read: if request.auth != null;
      allow update, delete: if request.auth != null;
    }
    
    // === ПУБЛИЧНЫЕ ДАННЫЕ ===
    match /public/{document=**} {
      allow read: if true;
      allow write: if request.auth != null;
    }
    
    // === ОСНОВНЫЕ ДАННЫЕ ===
    match /data/{document=**} {
      allow read: if request.auth != null;
      allow write: if request.auth != null;
    }
    
    // === НАСТРОЙКИ ===
    match /settings/{document=**} {
      allow read, write: if request.auth != null;
    }
    
    // === ALLOWLIST ===
    match /allowlist/{userId} {
      allow read: if request.auth != null;
      allow write: if request.auth != null;
    }
    
    // === ВХОДЯЩИЕ ЗАПРОСЫ ===
    match /inbox/{requestId} {
      allow create: if request.auth == null;
      allow read, write, delete: if request.auth != null;
    }
  }
}
